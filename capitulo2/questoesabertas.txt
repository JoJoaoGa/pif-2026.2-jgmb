#Questao 4
int a = 1, b = 2, c = 3, d = 4;

a += b + c;
= a = a + (b + c) → a = 1 + (2 + 3) = 6
a = 6

b *= c = d + 2;
c = 4 + 2 = 6
b = b * c -> b = 2 * 6 = 12
c = 6, b = 12

d -= c -= b -= a;
b = b - a -> b = 12 - 6 = 6
c = c - b -> c = 6 - 6 = 0
d = d - c -> d = 4 - 0 = 4
a = 6, b = 6, c = 0, d = 4

a += b += c += 7;
c = c + 7 -> c = 0 + 7 = 7
b = b + c -> b = 6 + 7 = 13
a = a + b -> a = 6 + 13 = 19
a = 19, b = 13, c = 7

d %= a + a + a;
a + a + a = 19 + 19 + 19 = 57
d = d % 57 -> d = 4 % 57 = 4
d = 4

Final: a = 19, b = 13, c = 7, d = 4
